134 void compile(const std::string kernel_source,
135 std::vector<std::string> names,
136 const bool add_reduction=false) {
137 std::ostringstream temp_stream;
138 temp_stream << reinterpret_cast<size_t> (this);
139 const std::string thread_id = temp_stream.str();
140
141 temp_stream.str(std::string());
142 temp_stream.clear();
143
144 temp_stream << "temp_" << thread_id << ".cpp";
145
146 const std::string filename = temp_stream.str();
147
148 if (jit::verbose) {
149 std::cout << "CPU info." << std::endl;
150 std::cout << " Command Line : " << std::endl;
151 }
152
153 char arg_string[] = CXX_ARGS;
154 llvm::SmallVector<const char *, 8> args = split_string(arg_string);
155 args.push_back(filename.c_str());
156#ifdef NDEBUG
157 args.push_back("-ffast-math");
158 args.push_back("-O3");
159#else
160 args.push_back("-debug-info-kind=standalone");
161#endif
162 if (jit::verbose) {
163 for (auto &arg : args) {
164 std::cout << " " << arg << std::endl;
165 }
166 }
167
168 clang::DiagnosticOptions diagnostic_options;
169 auto diagnostic_printer = std::make_unique<clang::TextDiagnosticPrinter> (llvm::errs(),
170 diagnostic_options);
171
172 auto diagnostic_ids = llvm::makeIntrusiveRefCnt<clang::DiagnosticIDs> ();
173 clang::DiagnosticsEngine diagnostic_engine(diagnostic_ids,
174 diagnostic_options,
175 diagnostic_printer.release());
176
177 auto invocation = std::make_shared<clang::CompilerInvocation> ();
178 clang::CompilerInvocation::CreateFromArgs(*(invocation.get()), args,
179 diagnostic_engine);
180
181 llvm::StringRef source_code_data(kernel_source);
182 auto buffer = llvm::MemoryBuffer::getMemBuffer(source_code_data);
183 invocation->getPreprocessorOpts().addRemappedFile(filename.c_str(),
184 buffer.release());
185
186 clang::CompilerInstance clang(invocation);
187 clang.createDiagnostics();
188
189 clang::TargetOptions target_options;
190 target_options.Triple = llvm::sys::getProcessTriple();
191 auto *target_info = clang::TargetInfo::CreateTargetInfo(diagnostic_engine,
192 target_options);
193 clang.setTarget(target_info);
194
195 clang::EmitLLVMOnlyAction action;
196 clang.ExecuteAction(action);
197
198 auto ir_module = action.takeModule();
199 auto context = std::unique_ptr<llvm::LLVMContext> (action.takeLLVMContext());
200
201 auto jit_try = llvm::orc::LLJITBuilder()
202#ifndef NDEBUG
203 .setPrePlatformSetup([](llvm::orc::LLJIT &J) {
204 return llvm::orc::enableDebuggerSupport(J);
205 })
206#endif
207 .create();
208 if (auto jiterror = jit_try.takeError()) {
209 std::cerr << "Failed to build JIT : " << toString(std::move(jiterror)) << std::endl;
210 exit(-1);
211 }
212 jit = std::move(jit_try.get());
213
214 auto error = jit->addIRModule(llvm::orc::ThreadSafeModule(std::move(ir_module),
215 llvm::orc::ThreadSafeContext(std::move(context))));
216
217#ifdef MACOS_LIB_RT
218 error = jit->linkStaticLibraryInto(jit->getMainJITDylib(), MACOS_LIB_RT);
219#endif
220 }

235 std::function<void(void)> create_kernel_call(const std::string kernel_name,
236 graph::input_nodes<T, SAFE_MATH> inputs,
237 graph::output_nodes<T, SAFE_MATH> outputs,
238 graph::input_nodes<T, SAFE_MATH> atomics,
239 graph::shared_random_state<T, SAFE_MATH> state,
240 const size_t num_rays,
241 const jit::texture1d_list &tex1d_list,
242 const jit::texture2d_list &tex2d_list) {
243 auto entry = std::move(jit->lookup(kernel_name)).get();
244
245 std::unordered_map<size_t, T *> buffers;
246
247 for (auto &input : inputs) {
248 if (!kernel_arguments.contains(input.get())) {
249 std::vector<T> arg(input->size());
250 memcpy(arg.data(), input->data(), input->size()*sizeof(T));
251 kernel_arguments[input.get()] = arg;
252 }
253 buffers[reinterpret_cast<size_t> (input.get())] = kernel_arguments[input.get()].data();
254 }
255 for (auto &output : outputs) {
256 if (!graph::atomic_accumulate_1D_cast(output).get()) {
257 if (!kernel_arguments.contains(output.get())) {
258 std::vector<T> arg(num_rays);
259 kernel_arguments[output.get()] = arg;
260 }
261 buffers[reinterpret_cast<size_t> (output.get())] = kernel_arguments[output.get()].data();
262 }
263 }
264 for (auto &atomic : atomics) {
265 if (!kernel_arguments.contains(atomic.get())) {
266 std::vector<T> arg(atomic->size());
267 memcpy(arg.data(), atomic->data(), atomic->size()*sizeof(T));
268 kernel_arguments[atomic.get()] = arg;
269 }
270 buffers[reinterpret_cast<size_t> (atomic.get())] = kernel_arguments[atomic.get()].data();
271 }
272
273 if (state.get()) {
274 auto kernel = entry.toPtr<void(*)(std::unordered_map<size_t, T *> &, typename graph::random_state_node<T, SAFE_MATH>::mt_state *)> ();
275
276 if (!kernel) {
277 std::cerr << "Failed to load function. " << kernel_name
278 << std::endl;
279 exit(-1);
280 }
281
282 if (jit::verbose) {
283 std::cout << " Function pointer: "
284 << reinterpret_cast<size_t> (kernel)
285 << std::endl;
286 }
287
288 return [kernel, buffers, state
289#ifdef PROFILE_KERNELS
290 , kernel_name
291#endif
292 ] () mutable {
293#ifdef PROFILE_KERNELS
294 timing::measure_diagnostic timer(kernel_name);
295#endif
296 kernel(buffers, state->data());
297#ifdef PROFILE_KERNELS
298 timer.print();
299#endif
300 };
301 } else {
302 auto kernel = entry.toPtr<void(*)(std::unordered_map<size_t, T *> &)> ();
303
304 if (!kernel) {
305 std::cerr << "Failed to load function. " << kernel_name
306 << std::endl;
307 exit(-1);
308 }
309
310 if (jit::verbose) {
311 std::cout << " Function pointer: "
312 << reinterpret_cast<size_t> (kernel)
313 << std::endl;
314 }
315
316 return [kernel, buffers
317#ifdef PROFILE_KERNELS
318 , kernel_name
319#endif
320 ] () mutable {
321#ifdef PROFILE_KERNELS
322 timing::measure_diagnostic timer(kernel_name);
323#endif
324 kernel(buffers);
325#ifdef PROFILE_KERNELS
326 timer.print();
327#endif
328 };
329 }
330 }

339 std::function<T(void)> create_max_call(graph::shared_leaf<T, SAFE_MATH> &argument,
340 std::function<void(void)> run) {
341 auto begin = kernel_arguments[argument.get()].cbegin();
342 auto end = kernel_arguments[argument.get()].cend();
343
344 return [run, begin, end] () mutable {
345 run();
346 if constexpr (jit::complex_scalar<T>) {
347 return *std::max_element(begin, end,
348 [] (const T a, const T b) {
349 return std::abs(a) < std::abs(b);
350 });
351 } else {
352 return *std::max_element(begin, end);
353 }
354 };

396 std::function<void(void)> create_copy_call(graph::copy_nodes<T, SAFE_MATH> &setters) {
397 std::vector<T *> sources;
398 std::vector<T *> destinations;
399 std::vector<size_t> sizes;
400
401 for (auto &[out, in] : setters) {
402 if (!kernel_arguments.contains(in.get())) {
403 std::vector<T> arg(in->size());
404 memcpy(arg.data(), in->data(), in->size()*sizeof(T));
405 kernel_arguments[in.get()] = arg;
406 }
407 destinations.push_back(kernel_arguments[in.get()].data());
408 sizes.push_back(in->size()*sizeof(T));
409
410 if (!kernel_arguments.contains(out.get())) {
411 std::vector<T> arg(out->size());
412 memcpy(arg.data(), out->data(), out->size()*sizeof(T));
413 kernel_arguments[out.get()] = arg;
414 }
415 sources.push_back(kernel_arguments[out.get()].data());
416 }
417
418 return [sources, destinations, sizes] () mutable {
419#ifdef PROFILE_KERNELS
420 timing::measure_diagnostic timer("copy buffer");
421#endif
422 for (size_t i = 0, ie = sources.size(); i < ie; i++) {
423 std::memcpy(destinations[i], sources[i], sizes[i]);
424 }
425#ifdef PROFILE_KERNELS
426 timer.print();
427#endif
