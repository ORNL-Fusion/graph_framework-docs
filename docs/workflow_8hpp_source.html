310 template<jit::float_scalar T, bool SAFE_MATH=false>
311 class converge_item final : public work_item<T, SAFE_MATH> {
312 private:
314 std::function<T(void)> max_kernel;
316 const T tolerance;
318 const size_t max_iterations;
319
320 public:
321//------------------------------------------------------------------------------
334//------------------------------------------------------------------------------
335 converge_item(graph::input_nodes<T, SAFE_MATH> &inputs,
336 graph::output_nodes<T, SAFE_MATH> &outputs,
337 graph::map_nodes<T, SAFE_MATH> &maps,
338 graph::input_nodes<T, SAFE_MATH> &atomics,
339 graph::shared_random_state<T, SAFE_MATH> &state,
340 const std::string name, const size_t size,
341 jit::context<T, SAFE_MATH> &context,
342 const T tol=1.0E-30,
343 const size_t max_iter=1000) :
344 work_item<T, SAFE_MATH> (inputs, outputs, maps, atomics,
345 state, name, size, context),
346 tolerance(tol), max_iterations(max_iter) {
347 context.add_max_reduction(size);
348 }
349
350//------------------------------------------------------------------------------
354//------------------------------------------------------------------------------
355 virtual void create_kernel_call(jit::context<T, SAFE_MATH> &context) {
356 work_item<T, SAFE_MATH>::create_kernel_call(context);
357 max_kernel = context.create_max_call(this->outputs.back(),
358 this->kernel);
359 }
360
361//------------------------------------------------------------------------------
363//------------------------------------------------------------------------------
364 virtual void run() {
365 size_t iterations = 0;
366 T max_residual = max_kernel();
367 T last_max = std::numeric_limits<T>::max();
368 T off_last_max = std::numeric_limits<T>::max();
369 while (std::abs(max_residual) > std::abs(tolerance) &&
370 std::abs(last_max - max_residual) > std::abs(tolerance) &&
371 std::abs(off_last_max - max_residual) > std::abs(tolerance) &&
372 iterations++ < max_iterations) {
373 last_max = max_residual;
374 if (!(iterations%2)) {
375 off_last_max = max_residual;
376 }
377 max_residual = max_kernel();
378 }
379
380// In release mode asserts are disables so write error to standard err. Need to
381// flip the comparison operator because we want to assert to trip if false.
382 assert(iterations < max_iterations &&
383 "Workitem failed to converge.");
384 if (iterations > max_iterations) {
385 std::cerr << "Workitem failed to converge with in given iterations."
386 << std::endl;
387 std::cerr << "Minimum residual reached: " << max_residual
388 << std::endl;
389 }
390 }
391 };
