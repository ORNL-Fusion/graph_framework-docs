411//------------------------------------------------------------------------------
415//------------------------------------------------------------------------------
416 virtual shared_leaf<T, SAFE_MATH> reduce() {
417// Constant reductions.
418 auto arg = constant_cast(this->arg);
419
420 if (arg.get()) {
421 return constant<T, SAFE_MATH> (this->evaluate());
422 }
423
424// !(a == b) -> a != b
425 auto equalc = equal_cast(this->arg);
426 if (equalc.get()) {
427 return equalc->get_left() != equalc->get_right();
428 }
429
430// !(a != b) -> a == b
431 auto nequalc = not_equal_cast(this->arg);
432 if (nequalc.get()) {
433 return nequalc->get_left() == nequalc->get_right();
434 }
435
436 if constexpr (!jit::complex_scalar<T>) {
437// !(a < b) -> a >= b
438 auto ltc = less_than_cast(this->arg);
439 if (ltc.get()) {
440 return ltc->get_left() >= ltc->get_right();
441 }
442
443// !(a <= b) -> a > b
444 auto lec = less_than_equal_cast(this->arg);
445 if (lec.get()) {
446 return lec->get_left() > lec->get_right();
447 }
448
449// !(a > b) -> a <= b
450 auto gtc = greater_than_cast(this->arg);
451 if (gtc.get()) {
452 return gtc->get_left() <= gtc->get_right();
453 }
454
455// !(a >= b) -> a < b
456 auto gec = greater_than_equal_cast(this->arg);
457 if (gec.get()) {
458 return gec->get_left() < gec->get_right();
459 }
460 }
461
462// !!a -> a
463 auto n = not_cast(this->arg);
464 if (n.get()) {
465 return n->get_arg();
466 }
467
468 return this->shared_from_this();
469 }
