574 if (size() > x.size()) {
575 assert(size()%x.size() == 0 &&
576 "Vector operand size is not a multiple of matrix operand size");
577
578 const size_t num_columns = size()/x.size();
579 const size_t num_rows = x.size();
580 for (size_t i = 0; i < num_rows; i++) {
581 for (size_t j = 0; j < num_columns; j++) {
582 if constexpr (jit::complex_scalar<T>) {
583 (*this)[i*num_columns + j] = std::atan(x[i]/(*this)[i*num_columns + j]);
584 } else {
585 (*this)[i*num_columns + j] = std::atan2(x[i], (*this)[i*num_columns + j]);
586 }
587 }
588 }

590 assert(x.size()%size() == 0 &&
591 "Vector operand size is not a multiple of matrix operand size");
592

596 for (size_t i = 0; i < num_rows; i++) {
597 for (size_t j = 0; j < num_columns; j++) {
