223 mt_state initialize_state(const uint32_t seed) {
224 mt_state state;
225 state.array[0] = seed;
226 for (uint16_t i = 1, ie = state.array.size(); i < ie; i++) {
227 state.array[i] = 1812433253U*(state.array[i - 1]^(state.array[i - 1] >> 30)) + i;
228 }
229 state.index = 0;
230
231 return state;

245 template<jit::float_scalar T, bool SAFE_MATH=false>
246 shared_leaf<T, SAFE_MATH> random_state(const size_t size,
247 const uint32_t seed=0) {
248 auto temp = std::make_shared<random_state_node<T, SAFE_MATH>> (size, seed)->reduce();
249// Test for hash collisions.
250 for (size_t i = temp->get_hash();
251 i < std::numeric_limits<size_t>::max(); i++) {
252 if (leaf_node<T, SAFE_MATH>::caches.nodes.find(i) ==
253 leaf_node<T, SAFE_MATH>::caches.nodes.end()) {
254 leaf_node<T, SAFE_MATH>::caches.nodes[i] = temp;
255 return temp;
256 } else if (temp->is_match(leaf_node<T, SAFE_MATH>::caches.nodes[i])) {
257 return leaf_node<T, SAFE_MATH>::caches.nodes[i];
258 }
259 }
260#if defined(__clang__) || defined(__GNUC__)
261 __builtin_unreachable();
262#else
263 assert(false && "Should never reach.");
264#endif
265 }
