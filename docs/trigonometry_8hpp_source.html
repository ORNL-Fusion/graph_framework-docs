64 virtual shared_leaf<T, SAFE_MATH> reduce() {
65 if (constant_cast(this->arg).get()) {
66 return constant<T, SAFE_MATH> (this->evaluate());
67 }
68
69 auto ap1 = piecewise_1D_cast(this->arg);
70 if (ap1.get()) {
71 return piecewise_1D(this->evaluate(), ap1->get_arg());
72 }
73
74 auto ap2 = piecewise_2D_cast(this->arg);
75 if (ap2.get()) {
76 return piecewise_2D(this->evaluate(),
77 ap2->get_num_columns(),
78 ap2->get_left(),
79 ap2->get_right());
80 }
81
82// Sin(ArcTan(x, y)) -> y/Sqrt(x^2 + y^2)
83 auto temp = atan_cast(this->arg);
84 if (temp.get()) {
85 return temp->get_right() /
86 (sqrt(temp->get_left()*temp->get_left() +
87 temp->get_right()*temp->get_right()));
88 }
89
90// Remove negative constants from the arguments.
91 auto am = multiply_cast(this->arg);
92 if (am.get()) {
93 auto lc = constant_cast(am->get_left());
94 if (lc.get() && lc->evaluate().is_negative()) {
95 return -sin(-this->arg);
96 }
97 }
98
99 return this->shared_from_this();
100 }

225 shared_leaf<T, SAFE_MATH> sin(shared_leaf<T, SAFE_MATH> x) {
226 auto temp = std::make_shared<sine_node<T, SAFE_MATH>> (x)->reduce();
227// Test for hash collisions.
228 for (size_t i = temp->get_hash(); i < std::numeric_limits<size_t>::max(); i++) {
229 if (leaf_node<T, SAFE_MATH>::caches.nodes.find(i) ==
230 leaf_node<T, SAFE_MATH>::caches.nodes.end()) {
231 leaf_node<T, SAFE_MATH>::caches.nodes[i] = temp;
232 return temp;
233 } else if (temp->is_match(leaf_node<T, SAFE_MATH>::caches.nodes[i])) {
234 return leaf_node<T, SAFE_MATH>::caches.nodes[i];
235 }
236 }
237#if defined(__clang__) || defined(__GNUC__)
238 __builtin_unreachable();
239#else
240 assert(false && "Should never reach.");
241#endif
242 }

312 auto c = constant_cast(this->arg);
313 if (c.get()) {
314// GNU complex revalues cos(0, i0) evaluates to (1, -i0). This is casuing a
315// problem with the c and fortran binding tests.
316 if (c->is(0)) {
317 return one<T, SAFE_MATH> ();
318 }
319 return constant<T, SAFE_MATH> (this->evaluate());
320 }
321
322 auto ap1 = piecewise_1D_cast(this->arg);
323 if (ap1.get()) {
324 return piecewise_1D(this->evaluate(),
325 ap1->get_arg());
326 }
327
328 auto ap2 = piecewise_2D_cast(this->arg);
329 if (ap2.get()) {
330 return piecewise_2D(this->evaluate(),
331 ap2->get_num_columns(),
332 ap2->get_left(),
333 ap2->get_right());
334 }
335
336// Cos(ArcTan(x, y)) -> x/Sqrt(x^2 + y^2)
337 auto temp = atan_cast(this->arg);
338 if (temp.get()) {
339 return temp->get_left() /
340 (sqrt(temp->get_left()*temp->get_left() +
341 temp->get_right()*temp->get_right()));
342 }
343
344// Remove negative constants from the arguments.
345 auto am = multiply_cast(this->arg);
346 if (am.get()) {
347 auto lc = constant_cast(am->get_left());
348 if (lc.get() && lc->evaluate().is_negative()) {
349 return cos(-this->arg);
350 }

353 return this->shared_from_this();

554 static std::string to_string(leaf_node<T, SAFE_MATH> *l,
555 leaf_node<T, SAFE_MATH> *r) {
556 return "atan" + jit::format_to_string(reinterpret_cast<size_t> (l))
557 + jit::format_to_string(reinterpret_cast<size_t> (r));
558 }

561//------------------------------------------------------------------------------
566//------------------------------------------------------------------------------
567 arctan_node(shared_leaf<T, SAFE_MATH> x,
568 shared_leaf<T, SAFE_MATH> y) :
569 branch_node<T, SAFE_MATH> (x, y, arctan_node::to_string(x.get(), y.get())) {}
570
571//------------------------------------------------------------------------------
577//------------------------------------------------------------------------------
578 virtual backend::buffer<T> evaluate() {
579 backend::buffer<T> left = this->left->evaluate();
580 backend::buffer<T> right = this->right->evaluate();
581 return backend::atan(left, right);
582 }

584//------------------------------------------------------------------------------
588//------------------------------------------------------------------------------
589 virtual shared_leaf<T, SAFE_MATH> reduce() {
590 auto l = constant_cast(this->left);
591 auto r = constant_cast(this->right);
592 if (l.get() && r.get()) {
593 return constant<T, SAFE_MATH> (this->evaluate());
594 }
595
596 auto pl1 = piecewise_1D_cast(this->left);
597 auto pr1 = piecewise_1D_cast(this->right);
598
599 if (pl1.get() && (r.get() || pl1->is_arg_match(this->right))) {
600 return piecewise_1D(this->evaluate(), pl1->get_arg());
601 } else if (pr1.get() && (l.get() || pr1->is_arg_match(this->left))) {
602 return piecewise_1D(this->evaluate(), pr1->get_arg());
603 }
604
605 auto pl2 = piecewise_2D_cast(this->left);
606 auto pr2 = piecewise_2D_cast(this->right);
607
608 if (pl2.get() && (r.get() || pl2->is_arg_match(this->right))) {
609 return piecewise_2D(this->evaluate(),
610 pl2->get_num_columns(),
611 pl2->get_left(),
612 pl2->get_right());
613 } else if (pr2.get() && (l.get() || pr2->is_arg_match(this->left))) {
614 return piecewise_2D(this->evaluate(),
615 pr2->get_num_columns(),
616 pr2->get_left(),
617 pr2->get_right());
618 }
619
620// Combine 2D and 1D piecewise constants if a row or column matches.
621 if (pr2.get() && pr2->is_row_match(this->left)) {
622 backend::buffer<T> result = pl1->evaluate();
623 result.atan_row(pr2->evaluate());
624 return piecewise_2D(result,
625 pr2->get_num_columns(),
626 pr2->get_left(),
627 pr2->get_right());
628 } else if (pr2.get() && pr2->is_col_match(this->left)) {
629 backend::buffer<T> result = pl1->evaluate();
630 result.atan_col(pr2->evaluate());
631 return piecewise_2D(result,
632 pr2->get_num_columns(),
633 pr2->get_left(),
634 pr2->get_right());
635 } else if (pl2.get() && pl2->is_row_match(this->right)) {
636 backend::buffer<T> result = pl2->evaluate();
637 result.atan_row(pr1->evaluate());
638 return piecewise_2D(result,
639 pl2->get_num_columns(),
640 pl2->get_left(),
641 pl2->get_right());
642 } else if (pl2.get() && pl2->is_col_match(this->right)) {
643 backend::buffer<T> result = pl2->evaluate();
644 result.atan_col(pr1->evaluate());
645 return piecewise_2D(result,
646 pl2->get_num_columns(),
647 pl2->get_left(),
648 pl2->get_right());
649 }
650
651 return this->shared_from_this();
652 }

661//------------------------------------------------------------------------------
662 virtual shared_leaf<T, SAFE_MATH>
663 df(shared_leaf<T, SAFE_MATH> x) {
664 if (this->is_match(x)) {
665 return one<T, SAFE_MATH> ();
666 }
667
668 const size_t hash = reinterpret_cast<size_t> (x.get());
669 if (this->df_cache.find(hash) == this->df_cache.end()) {
670 auto z = this->right/this->left;
671 this->df_cache[hash] = (1.0/(1.0 + z*z))*z->df(x);
672 }
673 return this->df_cache[hash];
674 }
