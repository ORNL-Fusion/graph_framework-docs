129//------------------------------------------------------------------------------
130 void add_kernel(const std::string name,
131 graph::input_nodes<T, SAFE_MATH> &inputs,
132 graph::output_nodes<T, SAFE_MATH> &outputs,
133 graph::map_nodes<T, SAFE_MATH> &setters,
134 graph::input_nodes<T, SAFE_MATH> &atomics,
135 graph::shared_random_state<T, SAFE_MATH> &state,
136 const size_t size,
137 const size_t iterations=1) {
138 kernel_names.push_back(name);
139
140 std::vector<bool> is_constant(inputs.size(), true);
141 visiter_map visited;
142 register_usage usage;
143 kernel_1dtextures[name] = texture1d_list();
144 kernel_2dtextures[name] = texture2d_list();
145 for (auto &[out, in] : setters) {
146 auto found = std::distance(inputs.begin(),
147 std::find(inputs.begin(),
148 inputs.end(), in));
149 if (found < is_constant.size()) {
150 is_constant[found] = false;
151 }
152 out->compile_preamble(source_buffer, registers,
153 visited, usage,
154 kernel_1dtextures[name],
155 kernel_2dtextures[name],
156 pre_funcs,
157 gpu_context.remaining_const_memory);
158 }
159 for (auto &out : outputs) {
160 out->compile_preamble(source_buffer, registers,
161 visited, usage,
162 kernel_1dtextures[name],
163 kernel_2dtextures[name],
164 pre_funcs,
165 gpu_context.remaining_const_memory);
166 }
167
168 for (auto &in : inputs) {
169 if (usage.find(in.get()) == usage.end()) {
170 usage[in.get()] = 0;
171 }
172 }
173
174 argument_set thread_shared;
175 jit::register_map thread_mem;
176
177 gpu_context.create_kernel_prefix(source_buffer,
178 name, inputs, outputs,
179 atomics, state,
180 size, is_constant,
181 registers, usage,
182 kernel_1dtextures[name],
183 kernel_2dtextures[name],
184 thread_shared,
185 thread_mem,
186 iterations);
187
188 for (auto &[out, in] : setters) {
189 out->compile(source_buffer, registers, thread_mem, usage);
190 }
191 for (auto &out : outputs) {
192 out->compile(source_buffer, registers, thread_mem, usage);
193 }
194
195 gpu_context.create_kernel_postfix(source_buffer, outputs,
196 setters, state,
197 registers, usage,
198 thread_shared,
199 thread_mem,
200 iterations);
201
202// Delete the registers so that they can be used again in other kernels.
203 std::vector<void *> removed_elements;
204 for (auto &[key, value] : registers) {
205 if (value[0] == 'r' ||
206 value[0] == 'l' ||
207 value[0] == 'i') {
208 removed_elements.push_back(key);
209 }
210 }
211 for (auto &out : outputs) {
212 if (graph::atomic_accumulate_1D_cast(out).get()) {
213 removed_elements.push_back(out.get());
214 }
215 }
216
217 for (auto &key : removed_elements) {
218 registers.erase(key);
219 }
220 }

264 filename << std::hash<std::string> {} (source)
265 << std::hash<std::thread::id>{}(std::this_thread::get_id());

268 } else if constexpr (use_metal<T> ()) {
269 filename << ".metal";
270 } else {
271 filename << ".cpp";
272 }
273
274 std::ofstream outFile(filename.str());
275 outFile << source;
