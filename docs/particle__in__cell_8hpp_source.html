1458//------------------------------------------------------------------------------
1459 template<std::floating_point T, model M>
1460 std::array<graph::shared_leaf<T>, 2> build_common_collision(const ion<T> &ion_a,
1461 const characteristics<T> &norms,
1462 graph::shared_leaf<T> uxb,
1463 graph::shared_leaf<T> tb,
1464 graph::shared_leaf<T> nb,
1465 const T mass_b,
1466 const uint8_t zb2,
1467 const T dt,
1468 const graph::shared_random_state<T> state) {
1469 const T mass_a = ion_a.mass*norms.m;
1470 const uint8_t za2 = ion_a.z*ion_a.z;
1471
1472// Convert to b frame.
1473 auto wxa = ion_a.v_para*norms.v - uxb;
1474 auto wya = ion_a.v_perp*norms.v;
1475
1476// Convert to sphereical.
1477 auto sphere = cartesian_to_sphereical(wxa, wya);
1478
1479// Apply Monte-Carlo collision operator.
1480 auto wtb = graph::sqrt(static_cast<T> (2)*q<T>*tb/mass_b);
1481 auto xab = sphere[0]/wtb;
1482
1483 auto erf_xab = graph::erf(xab);
1484 auto erfp_xab = erf_xab->df(xab);
1485 auto erfpp_xab = erfp_xab->df(xab);
1486
1487 auto gb = (erf_xab - xab*erfp_xab)/(static_cast<T> (2)*xab*xab);
1488 auto logA = static_cast<T> (30) - static_cast<T> (0.5)*graph::log(nb/(tb*graph::sqrt(tb)));
1489 auto nuab0 = nb*q<T>*q<T>*q<T>*q<T>*static_cast<T> (za2*zb2)*logA
1490 / (static_cast<T> (2)*std::numbers::pi_v<T>*mass_a*mass_a*epsilon0<T>*epsilon0<T>*wtb*wtb*wtb);
1491
1492// Velocity Scattering operator.
1493 auto nu_e_dt = build_colision_rate<T, M> (xab, mass_a, mass_b, gb, nuab0, erfp_xab)*dt;
1494 auto steps = graph::min(nu_e_dt*2.5, static_cast<T> (32));
1495
1496 nu_e_dt = nu_e_dt*steps;
1497
1498 auto E_nuE_d_nu_E_dE = static_cast<T> (0.5)*((static_cast<T> (3.0)*(xab*erfp_xab - erf_xab) - xab*xab*erfpp_xab)/(erf_xab - xab*erfp_xab));
1499
1500 const T mof = mass_a/(static_cast<T> (2)*q<T>);
1501 auto A = static_cast<T> (1) - static_cast<T> (2)*nu_e_dt;
1502 auto tbnu_e_df = tb*nu_e_dt;
1503 auto B = static_cast<T> (2)*tbnu_e_df*(static_cast<T> (1.5) + E_nuE_d_nu_E_dE);
1504
1505 auto rand1 = graph::random<T> (state);
1506 auto u_op = graph::apply_u(sphere[0]*sphere[0], steps, rand1,
1507 graph::constant(mof), tbnu_e_df, A, B);
1508 sphere[0] = graph::sqrt(u_op);
1509
1510 auto nu_D_dt = nuab0*(erf_xab - gb)/(xab*xab*xab)*dt;
1511 steps = graph::min(nu_D_dt, static_cast<T> (32));
1512 nu_D_dt = nu_D_dt/steps;
1513
1514 auto rand2 = graph::random<T> (state);
1515 sphere[1] = graph::apply_xi(sphere[1], steps, rand2, nu_D_dt);
1516
1517 sphere[1] = graph::if_(sphere[1]*sphere[1] > static_cast<T> (1),
1518 graph::copysign(static_cast<T> (1), sphere[1]) -
1519 sphere[1] % graph::copysign(static_cast<T> (1), sphere[1]),
1520 sphere[1]);
1521
1522 auto cart = sphereical_to_cartesian(sphere[0], sphere[1], sphere[2]);
1523 return {(cart[0] + uxb)/norms.v, cart[1]/norms.v};
1524 }
